custom_button = {#严苛税法
	name = EDG_LKSENG_GUI_impose_higher_taxes_button
	tooltip = EDG_LKSENG_GUI_impose_higher_taxes_button_tt
	potential = {
		FROM = {
			ai = no
			has_government_attribute = EDG_LKSENG_oppressive_taxation
		}
		owned_by = FROM
	}
	trigger = {
		base_manpower = 2
		NOT = {
			has_province_modifier = EDG_LKSENG_Recently_increased_taxes
		}
	}
	effect = {
		add_province_modifier = {
			name = EDG_LKSENG_Recently_increased_taxes
			duration = 1825
		}
		custom_tooltip = EDG_LKS_Recently_increased_taxes_tt
		hidden_effect = {
			set_variable = {
				EDG_LKS_Recently_increased_taxes_manpower_var = 0
			}
			export_to_variable = {
				which = EDG_LKS_Recently_increased_taxes_manpower_var
				value = trigger_value:base_manpower
			}
			subtract_variable = {
				which = EDG_LKS_Recently_increased_taxes_manpower_var
				value = 1
			}
			while = {
				limit = {
					check_variable = {
						which = EDG_LKS_Recently_increased_taxes_manpower_var
						value = 1
					}
				}
				subtract_variable = {
					which = EDG_LKS_Recently_increased_taxes_manpower_var
					value = 1
				}
				add_base_manpower = -1
				add_base_tax = 1
			}
			set_variable = {
				EDG_LKS_Recently_increased_taxes_tax_var = 0
			}
			export_to_variable = {
				which = EDG_LKS_Recently_increased_taxes_tax_var
				value = trigger_value:base_tax
			}
			multiply_variable = {
				which = EDG_LKS_Recently_increased_taxes_tax_var
				value = 25
			}
		}
	}
}
