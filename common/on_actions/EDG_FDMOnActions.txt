on_province_owner_change = {
	if = {
		limit = {
			province_id = 361
			owned_by = FDM
		}
		owner = {
			country_event = {
				id = EDG_FDMEvents.8
			}
		}
	}
	owner = {
		if = {
			limit = {
				ROOT = {
					is_city = yes
				}
				tag = FDM
				OR = {
					has_disaster = EDG_FDM_path_of_rebirth_disaster
					has_country_flag = EDG_FDM_path_of_rebirth_disaster_ended_flag_1
				}
			}
			if = {
				limit = {
					mission_completed = EDG_FDM_mission_1_4
				}
				add_treasury = 25
			}
			if = {
				limit = {
					mission_completed = EDG_FDM_mission_2_4
				}
				add_manpower = 0.5
			}
			if = {
				limit = {
					mission_completed = EDG_FDM_mission_3_4
				}
				add_adm_power = 5
			}
			if = {
				limit = {
					mission_completed = EDG_FDM_mission_4_4
				}
				add_dip_power = 5
			}
			if = {
				limit = {
					mission_completed = EDG_FDM_mission_5_4
				}
				add_mil_power = 5
			}
			if = {
				limit = {
					mission_completed = EDG_FDM_mission_2_5
				}
				add_prestige = 1
			}
			if = {
				limit = {
					mission_completed = EDG_FDM_mission_3_5
				}
				add_legitimacy = 1
			}
			if = {
				limit = {
					mission_completed = EDG_FDM_mission_4_5
				}
				add_splendor = 5
			}
			if = {
				limit = {
					mission_completed = EDG_FDM_mission_3_6
				}
				change_government_reform_progress = 2
			}
		}
	}
}

on_battle_won_country = {
	if = {
		limit = {
			mission_completed = EDG_FDM_mission_2_21
			NOT = {
				religion = FROM
			}
		}
		change_variable = {
			which = EDG_shiite_Caliph_authority_var
			value = 1
		}
	}
}

on_war_won = {
	if = {
		limit = {
			tag = FDM
			NOT = {
				religion = FROM
			}
		}
		change_variable = {
			which = EDG_FDM_war_won_heretic_and_heathen_var
			value = 1
		}

			limit = {
				mission_completed = EDG_FDM_mission_2_21
			}
			change_variable = {
				which = EDG_shiite_Caliph_authority_var
				value = 5
			}
		}
	}
}

on_core = {
	FROM = {
		if = {
			limit = {
				mission_completed = EDG_FDM_mission_3_9
			}
			change_variable = {
				which = EDG_shiite_Caliph_authority_var
				value = 1
			}
		}
		if = {
			limit = {
				mission_completed = EDG_FDM_mission_2_10
			}
			ROOT = {
				add_local_autonomy = -25
			}
		}
		if = {
			limit = {
				mission_completed = EDG_FDM_mission_3_10
			}
			ROOT = {
				add_nationalism = -10
			}
		}
		if = {
			limit = {
				mission_completed = EDG_FDM_mission_4_10
			}
			ROOT = {
				add_devastation = -10
			}
		}
	}
}
